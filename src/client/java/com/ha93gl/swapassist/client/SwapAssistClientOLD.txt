package com.ha93gl.swapassist.client;

import com.mojang.blaze3d.platform.InputConstants;
import net.fabricmc.api.ClientModInitializer;
import net.fabricmc.fabric.api.client.event.lifecycle.v1.ClientTickEvents;
import net.fabricmc.fabric.api.event.client.player.ClientPreAttackCallback;
import net.fabricmc.fabric.api.client.keymapping.v1.KeyMappingHelper;
import net.minecraft.client.KeyMapping;
import net.minecraft.client.Minecraft;
import net.minecraft.network.protocol.game.ServerboundSetCarriedItemPacket;
import net.minecraft.resources.Identifier;
import net.minecraft.world.entity.player.Inventory;

public class SwapAssistClient implements ClientModInitializer {

    private static final KeyMapping.Category CATEGORY =
            KeyMapping.Category.register(Identifier.parse("swap-assist-ha93gl:swap_assist"));

    private static KeyMapping swapAssistKey;
    private static KeyMapping swapModeKey;
    private static KeyMapping preselectSlotsKey;

    private static boolean swapAssistEnabled = false;
    private static SwapMode swapMode = SwapMode.DELAY_2_TICKS;

    private static int preselectSlot = Inventory.NOT_FOUND_INDEX;
    private static int attackSlot = Inventory.NOT_FOUND_INDEX;
    private static int pendingRestoreSlot = Inventory.NOT_FOUND_INDEX;
    private static int restoreDelay = 0;

    @Override
    public void onInitializeClient() {

        swapAssistKey = KeyMappingHelper.registerKeyMapping(new KeyMapping(
                "key.swap-assist-ha93gl.swap_assist",
                InputConstants.Type.KEYSYM,
                InputConstants.KEY_R,
                CATEGORY
        ));

        swapModeKey = KeyMappingHelper.registerKeyMapping(new KeyMapping(
                "key.swap-assist-ha93gl.swap_mode",
                InputConstants.Type.KEYSYM,
                InputConstants.KEY_G,
                CATEGORY
        ));

        preselectSlotsKey = KeyMappingHelper.registerKeyMapping(new KeyMapping(
                "key.swap-assist-ha93gl.preselect_slots",
                InputConstants.Type.KEYSYM,
                InputConstants.KEY_V,
                CATEGORY
        ));

        ClientTickEvents.END_CLIENT_TICK.register(client -> {
            while (swapAssistKey.consumeClick()) {
                swapAssistEnabled = !swapAssistEnabled;
            }

            while (swapModeKey.consumeClick()) {
                swapMode = swapMode.next();
                swapAssistEnabled = false;
            }

            while (preselectSlotsKey.consumeClick()) {
                if (client.player != null) {
                    preselectSlot = client.player.getInventory().getSelectedSlot();
                }
            }

            // Handle delayed restore
            if (pendingRestoreSlot != Inventory.NOT_FOUND_INDEX) {
                if (restoreDelay-- <= 0) {
                    if (client.player != null && client.getConnection() != null) {
                        client.player.getInventory().setSelectedSlot(pendingRestoreSlot);
                        client.getConnection().send(new ServerboundSetCarriedItemPacket(pendingRestoreSlot));
                    }
                    pendingRestoreSlot = Inventory.NOT_FOUND_INDEX;
                }
            }
        });

        ClientPreAttackCallback.EVENT.register((client, player, clickCount) -> {
            if (clickCount == 0 || !swapAssistEnabled || attackSlot == Inventory.NOT_FOUND_INDEX) {
                return false;
            }

            if (client.player == null || client.getConnection() == null) return false;

            int currentSlot = client.player.getInventory().getSelectedSlot();

            // Switch to attack slot
            client.player.getInventory().setSelectedSlot(attackSlot);
            client.getConnection().send(new ServerboundSetCarriedItemPacket(attackSlot));

            // Setup restore (unless mode is NO_RESTORE)
            if (swapMode != SwapMode.NO_RESTORE) {
                pendingRestoreSlot = currentSlot;
                restoreDelay = (swapMode == SwapMode.DELAY_2_TICKS) ? 2 : 0;
            }

            return false;
        });
    }

    private enum SwapMode {
        DELAY_2_TICKS,
        NO_RESTORE,
        FIXED_ATTRIBUTE;

        private SwapMode next() {
            return values()[(ordinal() + 1) % values().length];
        }
    }
}
